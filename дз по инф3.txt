def f2(a):
    return len(a)

def f3(a):
    count = 0
    for row in a:
        degree = 0
        for val in row:
            if val != 0:
                degree += 1
        if degree == 3:
            count += 1
    return count


def f4(a):
    count = 0
    for row in a:
        for val in row:
            if val != 0:
                count += 1
    return count // 2


def f5(a):
    total_sum = 0
    for row in a:
        for val in row:
            total_sum += val
    return total_sum // 2


def f6(a, vertex_index):
    count = 0
    for val in a[vertex_index]:
        if val != 0:
            count += 1
    return count

print("5. Сумма ветвей (весов) всех ребер:", f5(graph))
