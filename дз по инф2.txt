graph = [
    [0, 4, 0, 7, 0, 0, 0, 0],
    [4, 0, 4, 11, 0, 0, 0, 0],
    [0, 4, 0, 0, 0, 0, 0, 0],
    [7, 11, 4, 0, 0, 7, 0, 0],
    [0, 0, 0, 0, 0, 13, 13, 0],
    [0, 0, 0, 0, 13, 0, 13, 0],
    [0, 0, 0, 0, 13, 13, 0, 13],
    [0, 0, 0, 0, 0, 0, 13, 0]
]



def f2(a):
    return len(a)

def f3(a):
    count = 0
    for row in a:
        degree = 0
        for val in row:
            if val != 0:
                degree += 1
        if degree == 3:
            count += 1
    return count


def f4(a):
    count = 0
    for row in a:
        for val in row:
            if val != 0:
                count += 1
    return count // 2


def f5(a):
    total_sum = 0
    for row in a:
        for val in row:
            total_sum += val
    return total_sum // 2


def f6(a, vertex_index):
    count = 0
    for val in a[vertex_index]:
        if val != 0:
            count += 1
    return count


print("1. Матрица смежности:")

print("\n2. Количество вершин:", f2(graph))
print("3. Вершин со степенью 3:", f3(graph))
print("4. Количество ребер:", f4(graph))
print("5. Сумма ветвей (весов) всех ребер:", f5(graph))
print("6. Количество соседей у вершины M (индекс 6):", f6(graph, 6))
